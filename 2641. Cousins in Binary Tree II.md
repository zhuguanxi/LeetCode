## Problem Description
Given the root of a binary tree, replace the value of each node in the tree with the sum of all its cousins' values.

Two nodes of a binary tree are cousins if they have the same depth with different parents.

Return the root of the modified tree.

Note that the depth of a node is the number of edges in the path from the root node to it.

### WAY 1:
* 執行兩次 DFS，第一次先算 level sum，第二次再改值
```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void level_sum(TreeNode* node, int depth, vector<int>& levelSum)
    {
        if (node == nullptr) return;

        if (depth >= levelSum.size())
            levelSum.push_back(0);

        levelSum[depth]+=node->val;

        if (node->left) level_sum(node->left, depth+1, levelSum);
        if (node->right) level_sum(node->right, depth+1, levelSum);
    }
    void update_sum(TreeNode* node, int depth, vector<int>& levelSum)
    {
        if (node == nullptr) return;

        int sum = 0;
        if (node->left) sum+=node->left->val;
        if (node->right) sum+=node->right->val;

        if (node->left) node->left->val = levelSum[depth+1] - sum;
        if (node->right) node->right->val = levelSum[depth+1] - sum;

        update_sum(node->left, depth+1, levelSum);
        update_sum(node->right, depth+1, levelSum);

    }
    TreeNode* replaceValueInTree(TreeNode* root) {
        vector<int> levelSum;
        level_sum(root, 0, levelSum);

        root->val = 0;

        update_sum(root, 0, levelSum);

        return root;
    }
};
```
* n：number of nodes ([1, 10e5])
* h : 樹高
* Time Complexity $O(n)$
* Space Complexity $O(h)$
