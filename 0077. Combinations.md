## Problem Description
Given two integers n and k, return all possible combinations of k numbers chosen from the range [1, n].

You may return the answer in any order.

### WAY 1:
* 回朔法，還可以加入條件來優化速度
```
class Solution {
public:
    void dfs(int n, vector<vector<int>>& ans, vector<int>& path,  int k)
    {
        
        if (path.size() == k)
        {
            ans.push_back(path);
            return ;
        }
        for (int i=n; i>0; i--)
        {
            path.push_back(i);
            dfs(i-1, ans, path, k);
            path.pop_back();

        }
    }
    vector<vector<int>> combine(int n, int k) {
        vector<vector<int>> ans;
        vector<int> path;
        dfs(n, ans, path, k);
        
        return ans;
    }
};
```
